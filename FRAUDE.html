import pandas as pd from IPython.display import display # 1. Données df = pd.DataFrame({ "Zone": [ "Maison 1", "Maison 2", "Maison 3", "Maison 4", "Maison 5", "Maison 6", "Maison 7", "Maison 8", "Maison 9", "Maison 10", "Office National de l'Artisanat" ], "Consommation Juillet": [ 150, 100, 100, 100, 150, 100, 100, 100, 100, 150, 950 ], "Consommation Aout": [ 150, 100, 100, 100, 150, 100, 100, 100, 0, 150, 950 ] }) # 2. Calculs par zone df["Ecart (Juillet - Aout)"] = ( df["Consommation Juillet"] - df["Consommation Aout"] ) df["Variation (%)"] = ( (df["Consommation Aout"] - df["Consommation Juillet"]) / df["Consommation Juillet"].replace(0, float("nan")) * 100 ).round(2) def diagnostic(row): if row["Consommation Juillet"] > 0 and row["Consommation Aout"] == 0: return "ALERTE : à vérifier" elif row["Consommation Aout"] < row["Consommation Juillet"]: return "Baisse à vérifier" elif row["Consommation Aout"] > row["Consommation Juillet"]: return "Hausse à vérifier" return "Stable" df["Diagnostic"] = df.apply(diagnostic, axis=1) # 3. Ajouter la ligne Total total_juillet = df["Consommation Juillet"].sum() total_aout = df["Consommation Aout"].sum() total = pd.DataFrame([{ "Zone": "TOTAL DES ZONES", "Consommation Juillet": total_juillet, "Consommation Aout": total_aout, "Ecart (Juillet - Aout)": total_juillet - total_aout, "Variation (%)": round( (total_aout - total_juillet) / total_juillet * 100, 2 ), "Diagnostic": "Totaux recalculés" }]) resultat = pd.concat([df, total], ignore_index=True) # 4. Tableau de vérification des compteurs généraux compteurs = pd.DataFrame({ "Mois": ["Juillet", "Août"], "Total des zones recalculé": [total_juillet, total_aout], "Compteur général": [2000, 2000] }) compteurs["Ecart compteur - zones"] = ( compteurs["Compteur général"] - compteurs["Total des zones recalculé"] ) compteurs["Statut"] = compteurs["Ecart compteur - zones"].apply( lambda x: "Écart à vérifier" if x != 0 else "Conforme" ) # 5. Affichage organisé print("TABLEAU 1 : ANALYSE DE CHAQUE ZONE") display(resultat.style .format({ "Consommation Juillet": "{:.0f}", "Consommation Aout": "{:.0f}", "Ecart (Juillet - Aout)": "{:.0f}", "Variation (%)": "{:.2f}%" }) .background_gradient(subset=["Ecart (Juillet - Aout)"], cmap="RdYlGn") .set_properties(**{"text-align": "center"}) .set_caption("Analyse mensuelle des consommations")) print("TABLEAU 2 : CONTROLE DES COMPTEURS GENERAUX") display(compteurs.style .format({ "Total des zones recalculé": "{:.0f}", "Compteur général": "{:.0f}", "Ecart compteur - zones": "{:.0f}" }) .set_properties(**{"text-align": "center"}) .set_caption("Vérification des écarts")) print("CONCLUSION") print(f"Total recalculé en juillet : {total_juillet}") print(f"Total recalculé en août : {total_aout}") print("Maison à vérifier en priorité : Maison 9 (consommation nulle en août).") print("Un écart ou une baisse ne prouve pas à lui seul une fraude.")
